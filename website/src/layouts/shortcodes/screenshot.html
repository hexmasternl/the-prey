{{- /* An app screenshot in the page's language: {{< screenshot name="area-editor" alt="…" caption="…" >}} renders static/images/screens/{lang}/area-editor-thumb.webp as a small image that links to the full-size static/images/screens/{lang}/area-editor.webp; main.js opens that link in the lightbox. The English and Dutch twins of a page share the same call and each shows the app in its own language. A missing capture renders nothing and logs a build warning, so `hugo` output must be free of "screenshot" warnings before a release. */ -}} {{- $name := .Get "name" -}} {{- $path := printf "images/screens/%s/%s.webp" .Page.Language.Lang $name -}} {{- $thumb := printf "images/screens/%s/%s-thumb.webp" .Page.Language.Lang $name -}} {{- $file := printf "static/%s" $path -}} {{- $thumbFile := printf "static/%s" $thumb -}} {{- if not (and (fileExists $file) (fileExists $thumbFile)) -}} {{- warnf "screenshot %q: %s or its -thumb does not exist (%s)" $name $file .Position -}} {{- else -}} {{- $img := imageConfig $thumbFile -}} {{- end -}}
